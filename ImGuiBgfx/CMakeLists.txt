cmake_minimum_required(VERSION 3.25)
project(ImGuiBgfx LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The SDL demo currently supports Windows/MSVC.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../Dependencies/Packages")
find_package(SDL3 CONFIG REQUIRED COMPONENTS SDL3-static)
find_package(bgfx CONFIG REQUIRED)

add_library(imgui STATIC
  Source/ImGui/imgui.cpp Source/ImGui/imgui_draw.cpp Source/ImGui/imgui_tables.cpp
  Source/ImGui/imgui_widgets.cpp Source/ImGui/imgui_demo.cpp)
target_include_directories(imgui PUBLIC Source/ImGui)
add_library(imgui_bgfx STATIC Source/ImGui/imgui_impl_bgfx.cpp)
target_link_libraries(imgui_bgfx PUBLIC imgui bgfx::bgfx)
# Consume shader binaries from the same checkout used to build bgfx. No shaderc needed.
set(BGFX_IMGUI_SHADER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Dependencies/Repos/BgfxBuild/bgfx/examples/common/imgui")
if(NOT EXISTS "${BGFX_IMGUI_SHADER_DIR}/vs_ocornut_imgui.bin.h" OR NOT EXISTS "${BGFX_IMGUI_SHADER_DIR}/fs_ocornut_imgui.bin.h")
  message(FATAL_ERROR "Missing bgfx ImGui shader binaries in ${BGFX_IMGUI_SHADER_DIR}")
endif()
target_include_directories(imgui_bgfx PRIVATE "${BGFX_IMGUI_SHADER_DIR}")
add_executable(ImGuiBgfx Source/main.cpp Source/ImGui/imgui_impl_sdl3.cpp)
target_compile_definitions(ImGuiBgfx PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(ImGuiBgfx PRIVATE imgui_bgfx SDL3::SDL3-static)
foreach(target imgui imgui_bgfx ImGuiBgfx)
  target_compile_options(${target} PRIVATE /utf-8 /W4)
  target_compile_definitions(${target} PRIVATE NOMINMAX)
endforeach()

include(CTest)
if(BUILD_TESTING)
  add_executable(ImGuiBgfxBackendTests Tests/backend_tests.cpp)
  target_link_libraries(ImGuiBgfxBackendTests PRIVATE imgui_bgfx)
  target_compile_options(ImGuiBgfxBackendTests PRIVATE /utf-8 /W4)
  target_compile_definitions(ImGuiBgfxBackendTests PRIVATE NOMINMAX)
  add_test(NAME ImGuiBgfxBackend COMMAND ImGuiBgfxBackendTests)
  set_tests_properties(ImGuiBgfxBackend PROPERTIES TIMEOUT 45)
endif()
